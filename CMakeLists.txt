cmake_minimum_required(VERSION 3.21)
project(sdf_budget_lab VERSION 0.1.0 LANGUAGES CXX)
option(SDF_WITH_CGAL "Build exact mesh-distance reference" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(sdf_core src/field.cpp src/oracle.cpp src/metrics.cpp src/allocation.cpp src/precision.cpp src/surface.cpp src/tail.cpp)
target_include_directories(sdf_core PUBLIC include)
find_path(SDF_BOOST_HEADERS boost/multiprecision/cpp_int.hpp
  HINTS "${CMAKE_CURRENT_SOURCE_DIR}/.deps/vcpkg/installed/x64-windows/include"
        "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/include")
if(NOT SDF_BOOST_HEADERS)
  message(FATAL_ERROR "Compact mesh exact predicates require Boost.Multiprecision headers")
endif()
add_library(sdf_compact src/compact_mesh.cpp src/compact_validation.cpp src/compact_surface.cpp)
target_include_directories(sdf_compact PUBLIC include PRIVATE "${SDF_BOOST_HEADERS}")
target_link_libraries(sdf_compact PUBLIC sdf_core)
add_executable(sdf_compact_bench src/compact_benchmark.cpp)
target_link_libraries(sdf_compact_bench PRIVATE sdf_compact)
if(WIN32)
  target_link_libraries(sdf_compact_bench PRIVATE psapi)
endif()
if(MSVC)
  target_compile_options(sdf_compact PRIVATE /W4 /permissive- /fp:strict)
else()
  target_compile_options(sdf_compact PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
if(MSVC)
  target_compile_options(sdf_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(sdf_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(SDF_WITH_CGAL)
  find_package(CGAL 5.6 REQUIRED)
  target_sources(sdf_core PRIVATE src/mesh_oracle.cpp src/mesh_oracle_fast.cpp)
  target_link_libraries(sdf_core PUBLIC CGAL::CGAL)
  target_compile_definitions(sdf_core PUBLIC SDF_WITH_CGAL=1)
endif()
add_executable(sdf_bench src/benchmark.cpp)
target_link_libraries(sdf_bench PRIVATE sdf_core)
target_compile_definitions(sdf_bench PRIVATE SDF_BUILD_CONFIGURATION="$<CONFIG>")
add_executable(sdf_allocation_bench src/allocation_benchmark.cpp)
target_link_libraries(sdf_allocation_bench PRIVATE sdf_core)
target_compile_definitions(sdf_allocation_bench PRIVATE SDF_BUILD_CONFIGURATION="$<CONFIG>")
add_executable(sdf_precision_bench src/precision_benchmark.cpp)
target_link_libraries(sdf_precision_bench PRIVATE sdf_core)
target_compile_definitions(sdf_precision_bench PRIVATE SDF_BUILD_CONFIGURATION="$<CONFIG>")
add_executable(sdf_tail_bench src/tail_benchmark.cpp)
target_link_libraries(sdf_tail_bench PRIVATE sdf_core)
target_compile_definitions(sdf_tail_bench PRIVATE SDF_BUILD_CONFIGURATION="$<CONFIG>")
include(CTest)
add_executable(sdf_distortion_bench src/distortion_benchmark.cpp)
target_link_libraries(sdf_distortion_bench PRIVATE sdf_core)
target_compile_definitions(sdf_distortion_bench PRIVATE SDF_BUILD_CONFIGURATION="$<CONFIG>")
add_executable(sdf_refinement_bench src/refinement_benchmark.cpp)
target_link_libraries(sdf_refinement_bench PRIVATE sdf_core)
target_compile_definitions(sdf_refinement_bench PRIVATE SDF_BUILD_CONFIGURATION="$<CONFIG>")
if(BUILD_TESTING)
  add_executable(sdf_tests tests/test_core.cpp)
  target_link_libraries(sdf_tests PRIVATE sdf_core)
  add_test(NAME sdf_correctness COMMAND sdf_tests ${CMAKE_CURRENT_SOURCE_DIR}/fixtures)
  add_executable(sdf_allocation_tests tests/test_allocation.cpp)
  target_link_libraries(sdf_allocation_tests PRIVATE sdf_core)
  add_test(NAME sdf_allocation_correctness COMMAND sdf_allocation_tests)
  add_executable(sdf_precision_tests tests/test_precision.cpp)
  target_link_libraries(sdf_precision_tests PRIVATE sdf_core)
  add_test(NAME sdf_precision_correctness COMMAND sdf_precision_tests)
  add_executable(sdf_tail_tests tests/test_tail.cpp)
  target_link_libraries(sdf_tail_tests PRIVATE sdf_core)
  add_test(NAME sdf_tail_correctness COMMAND sdf_tail_tests)
endif()
add_executable(sdf_fidelity_bench src/fidelity_benchmark.cpp)
target_link_libraries(sdf_fidelity_bench PRIVATE sdf_core)
target_compile_definitions(sdf_fidelity_bench PRIVATE SDF_BUILD_CONFIGURATION="$<CONFIG>")

add_executable(sdf_generalization_bench src/generalization_benchmark.cpp)
target_link_libraries(sdf_generalization_bench PRIVATE sdf_core)
target_compile_definitions(sdf_generalization_bench PRIVATE SDF_BUILD_CONFIGURATION="$<CONFIG>")
add_executable(sdf_generalization_v2_diagnostic src/generalization_v2_diagnostic.cpp)
add_executable(sdf_generalization_v2_baseline src/generalization_v2_baseline.cpp)
target_link_libraries(sdf_generalization_v2_baseline PRIVATE sdf_core)
target_compile_definitions(sdf_generalization_v2_baseline PRIVATE SDF_BUILD_CONFIGURATION="$<CONFIG>")
target_link_libraries(sdf_generalization_v2_diagnostic PRIVATE sdf_core)
if(WIN32)
  target_link_libraries(sdf_generalization_v2_diagnostic PRIVATE psapi)
endif()
if(BUILD_TESTING)
  add_executable(sdf_generalization_tests tests/test_generalization.cpp)
  target_link_libraries(sdf_generalization_tests PRIVATE sdf_core)
  add_test(NAME sdf_generalization_correctness COMMAND sdf_generalization_tests)
  add_executable(sdf_ray_tests tests/test_ray.cpp)
  target_link_libraries(sdf_ray_tests PRIVATE sdf_core)
  add_test(NAME sdf_segmented_ray_correctness COMMAND sdf_ray_tests)
  add_executable(sdf_compact_tests tests/test_compact.cpp)
  target_link_libraries(sdf_compact_tests PRIVATE sdf_compact)
  add_test(NAME sdf_compact_mesh_correctness COMMAND sdf_compact_tests)
endif()
